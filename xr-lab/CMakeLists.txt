cmake_minimum_required(VERSION 3.22)
project(qss_xr_lab LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenXR REQUIRED CONFIG)
add_library(qss_native_glue STATIC "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
set_target_properties(qss_native_glue PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(qss_native_glue PUBLIC "${ANDROID_NDK}/sources/android/native_app_glue")
add_library(qss_xr_lab SHARED src/main/cpp/xr_host.cpp)
target_include_directories(qss_xr_lab PRIVATE include)
target_compile_definitions(qss_xr_lab PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
target_link_options(qss_xr_lab PRIVATE "-uANativeActivity_onCreate")
target_link_libraries(qss_xr_lab PRIVATE OpenXR::openxr_loader qss_native_glue android log EGL GLESv3)
